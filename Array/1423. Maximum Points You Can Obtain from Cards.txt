class Solution {
public:
    int maxScore(vector<int>& cardPoints, int k) {
        int maxf=0,sum=0,lSum=0,rSum=0;
        int n=cardPoints.size()-1;
        for(int i=0;i<k;i++) {
            lSum=lSum+cardPoints[i];
        }
        maxf=lSum;
        int rightIndex=n;
        for(int i=k-1;i>=0;i--) {
            lSum=lSum-cardPoints[i];
            rSum=rSum+cardPoints[rightIndex];
            sum=lSum+rSum;
            rightIndex--;
            maxf=max(maxf,sum);
        }
        return maxf;
    }
};
