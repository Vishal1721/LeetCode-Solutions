class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        int i=0,j=0;
        unordered_map<int,int>st;
        int cnt=0,maxf=0;
        while(i<fruits.size() && j<fruits.size()) {
            if(st.find(fruits[j])==st.end()) {
                cnt++;
            }
            st[fruits[j]]++;
            while(cnt>2) {
             
                if(st.find(fruits[i])!=st.end()) {
                    st[fruits[i]]--;    
                    if(st[fruits[i]]==0){
                        cnt--;
                        st.erase(fruits[i]);
                    }
                }
                i++;
            }
            maxf=max(maxf,j-i+1);
            j++;
        }
        return maxf;
    }
};
