class Solution {
public:
    int numberOfSubarrays(vector<int>& nums, int k) {
         unordered_map<int,int>mp;
        int prefSum=0,cnt=0;
        mp[0]=1;
        for(int i=0;i<nums.size();i++) {
            if(nums[i]%2==1) nums[i]=1;
            else nums[i]=0;
        }
        for(int i=0;i<nums.size();i++) {
            prefSum+=nums[i];
            int ans=prefSum-k;
            if(mp.find(ans)!=mp.end()) {
                cnt+=mp[ans];
            }
            mp[prefSum]++;
        }
        return cnt;
    }
};
